/**
 * מסלול ה-AI — קליטת "שליחה למורה" לגיליון Google Sheets.
 *
 * איך מתקינים (חד פעמי):
 * 1. sheets.google.com -> גיליון חדש וריק. תני לו שם כמו "מסלול AI - תשובות תלמידים".
 * 2. בתפריט: Extensions (תוספים) -> Apps Script.
 * 3. תמחקי את כל הקוד שכבר שם (myFunction הריק), ותדביקי את כל הקובץ הזה במקומו.
 * 4. Deploy (פריסה) -> New deployment (פריסה חדשה) -> לחצי על גלגל השיניים ליד "Select type" ובחרי "Web app".
 * 5. Execute as: Me (את). Who has access: Anyone.
 *    (הערה: "Anyone" אומר שכל מי שיש לו את הקישור יכול לשלוח נתונים - זה בסדר לכלי כיתתי,
 *    אבל אם תרצי הגבלה נוספת בעתיד אפשר לדבר על זה)
 * 6. Deploy -> תאשרי הרשאות (זה הגיליון שלך, זה בטוח) -> תעתיקי את ה-Web app URL שמופיע בסוף.
 * 7. תשלחי לי את ה-URL הזה ואני אחבר אותו לדפים.
 *
 * עדכון (הסטודיו, 10.2026): הסקריפט קולט גם משחקים (קובצי HTML) וגם כוחות מדף הסטודיו.
 * איך מעדכנים בלי לשנות את הקישור:
 * 1. Apps Script -> מוחקים את הקוד הישן ומדביקים את כל הקובץ הזה.
 * 2. Deploy -> Manage deployments (ניהול פריסות) -> העיפרון (Edit) -> Version: New version -> Deploy.
 *    (לא New deployment! אחרת נוצר קישור חדש והדפים לא יעבדו)
 * 3. גוגל יבקש הרשאה חדשה לדרייב. זה כדי לשמור את קובצי המשחקים בתיקייה בדרייב שלך.
 * המשחקים נשמרים בתיקייה "ארקייד הסטודיו - משחקים שהוגשו", והרשימה בלשונית "ארקייד" בגיליון.
 *
 * עדכון (שמירה בענן, 13.10.2026): לשונית "התקדמות", שורה אחת לכל כינוי, עם הקוד הסודי.
 * מעדכנים באותה דרך (Manage deployments -> New version). אין הרשאה חדשה.
 * תלמיד ששכח את הקוד: מסתכלים בעמודה "קוד סודי". אפשר גם לשנות אותו שם ל-4 ספרות אחרות.
 * את עמודת "נתונים" לא עורכים ידנית. משם הסטודיו טוען את ההתקדמות.
 *
 * עדכון (אישור מהגיליון + קיר הרעיונות, 21.10.2026):
 * - מאשרים בגיליון: בעמודת "סטטוס" בוחרים "מאושר", ותוך דקה הפריט מופיע באתר לבד.
 *   כוחות (לשונית "כוחות") עולים ללוח הכוחות, משחקים (לשונית "ארקייד") לארקייד, רעיונות (לשונית "רעיונות") לקיר הרעיונות.
 *   "לא עכשיו" או כל דבר אחר = לא מופיע. באתר מופיע רק הכינוי, אף פעם לא השם האמיתי.
 * - לשונית "הצטרפויות": מי רוצה להשתתף בהאקתון, ועל איזה רעיון. זו רשימת המשתתפים.
 * מעדכנים באותה דרך (Manage deployments -> העיפרון -> New version -> Deploy). אין הרשאה חדשה.
 */

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("תשובות") || ss.insertSheet("תשובות");

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["זמן שליחה", "שם תלמיד/ה", "מסלול", "תחנה", "שאלה", "תשובה", "תאריך (מהתלמיד)", "עבד/ה בזוג עם", "מזהה שליחה"]);
    sheet.setFrozenRows(1);
  }

  var data = JSON.parse(e.postData.contents);
  // הסטודיו: משחק או כוח. שליחה רגילה מדף תחנה ממשיכה כמו קודם
  if (data.kind === "game") return saveGame_(ss, data);
  if (data.kind === "power") return savePower_(ss, data);
  if (data.kind === "cloud") return cloud_(ss, data);
  // קיר הרעיונות
  if (data.kind === "idea") return saveIdea_(ss, data);
  if (data.kind === "join") return saveJoin_(ss, data);

  var trackLabel = data.track === "tichon" ? "תיכון" : "חטיבה";

  (data.answers || []).forEach(function (a) {
    sheet.appendRow([
      data.timestamp, data.name, trackLabel, data.pageTitle,
      a.question, a.answer,
      data.date, data.partner, data.submissionId
    ]);
  });

  return ContentService
    .createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ---------- הסטודיו ----------

function ok_() {
  return ContentService
    .createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}

// גיליון עם שורת כותרת, נוצר בפעם הראשונה
function sheetWithHeader_(ss, name, header) {
  var sh = ss.getSheetByName(name) || ss.insertSheet(name);
  if (sh.getLastRow() === 0) {
    sh.appendRow(header);
    sh.setFrozenRows(1);
  }
  statusMenu_(sh);
  return sh;
}

// תפריט נפתח בעמודת "סטטוס" (פעם אחת לכל לשונית). אפשר גם להקליד ידנית
var STATUS_LIST = ["חדש", "ממתין לאישור", "מאושר", "לא עכשיו"];
function statusMenu_(sh) {
  var head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
  var col = head.indexOf("סטטוס") + 1;
  if (!col || sh.getMaxRows() < 2) return;
  var cell = sh.getRange(2, col);
  if (cell.getDataValidation()) return;
  var rule = SpreadsheetApp.newDataValidation().requireValueInList(STATUS_LIST, true).setAllowInvalid(true).build();
  sh.getRange(2, col, sh.getMaxRows() - 1, 1).setDataValidation(rule);
}

function clean_(v, max) {
  return String(v || "").replace(/[\r\n\t]+/g, " ").trim().slice(0, max || 200);
}

// שומר את קובץ ה-HTML בתיקייה בדרייב, ושורה בלשונית "ארקייד". לא מתפרסם לשום מקום בלי אישור המורה
function saveGame_(ss, data) {
  var html = String(data.html || "");
  if (!html || html.length > 5000000) return ok_();

  var folderName = "ארקייד הסטודיו - משחקים שהוגשו";
  var it = DriveApp.getFoldersByName(folderName);
  var folder = it.hasNext() ? it.next() : DriveApp.createFolder(folderName);

  var safeTitle = clean_(data.gameName, 40).replace(/[\\\/:*?"<>|]/g, "") || "משחק";
  var stamp = Utilities.formatDate(new Date(), "Asia/Jerusalem", "yyyy-MM-dd HH-mm");
  var file = folder.createFile(stamp + " - " + clean_(data.nick, 30) + " - " + safeTitle + ".html", html, MimeType.HTML);

  var sh = sheetWithHeader_(ss, "ארקייד",
    ["זמן", "שם (למורה)", "כינוי בארקייד", "שם המשחק", "ענף", "במה אני הכי גאה", "קובץ", "סטטוס"]);
  sh.appendRow([
    new Date(), clean_(data.realName, 60), clean_(data.nick, 30), clean_(data.gameName, 60),
    clean_(data.genre, 40), clean_(data.proud, 300), file.getUrl(), "ממתין לאישור"
  ]);
  return ok_();
}

// כוח חדש שתלמיד גילה. המורה מחליטה אם הוא עולה ללוח
function savePower_(ss, data) {
  var sh = sheetWithHeader_(ss, "כוחות",
    ["זמן", "שם (למורה)", "כינוי", "שם הכוח", "מתי משתמשים", "הלחש", "סטטוס"]);
  sh.appendRow([
    new Date(), clean_(data.realName, 60), clean_(data.nick, 30), clean_(data.powerName, 60),
    clean_(data.when, 300), clean_(data.spell, 500), "חדש"
  ]);
  return ok_();
}

// ---------- שמירה בענן (הסטודיו) ----------
// לשונית "התקדמות": שורה אחת לכל כינוי. הקוד גלוי למורה, כדי שתוכל לעזור למי ששכח
var CLOUD_HEAD = ["כינוי", "שם (למורה)", "קוד סודי", "דמות", "נקודות", "דרגה", "אתגרים", "כוחות", "סודות", "משחקים שנשלחו", "נוצר", "שמירה אחרונה", "נתונים (לא לערוך)"];

function json_(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

// "נינג׳ה  הפיקסלים" ו"נינג'ה הפיקסלים" הם אותו כינוי
function nickKey_(v) {
  return String(v || "").replace(/[׳'`’]/g, "'").replace(/[״"”]/g, '"').replace(/\s+/g, " ").trim().toLowerCase();
}

// כינוי שמתחיל ב-= לא יהפוך לנוסחה בגיליון
function text_(v) {
  return /^[=+\-@]/.test(v) ? "'" + v : v;
}

// op: register (פעם ראשונה), save (שמירה אוטומטית), load (כניסה ממחשב אחר)
function cloud_(ss, data) {
  var nick = clean_(data.nick, 20).replace(/\s+/g, " "), code = String(data.code || "");
  if (!nick || !/^\d{4}$/.test(code)) return json_({ cloud: 1, ok: false, err: "bad" });

  // הגנה מניחוש קודים: 8 טעויות על אותו כינוי -> רבע שעה המתנה
  var cache = CacheService.getScriptCache(), failKey = "fail:" + nickKey_(nick);
  var fails = +(cache.get(failKey) || 0);
  if (fails >= 8) return json_({ cloud: 1, ok: false, err: "wait" });

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var sh = ss.getSheetByName("התקדמות");
    if (!sh) {
      sh = ss.insertSheet("התקדמות");
      sh.appendRow(CLOUD_HEAD);
      sh.setFrozenRows(1);
      sh.getRange("C:C").setNumberFormat("@"); // שהקוד 0123 לא יהפוך ל-123
    }
    var last = sh.getLastRow(), row = 0, key = nickKey_(nick);
    if (last > 1) {
      var nicks = sh.getRange(2, 1, last - 1, 1).getDisplayValues();
      for (var i = 0; i < nicks.length; i++) if (nickKey_(nicks[i][0]) === key) { row = i + 2; break; }
    }

    if (!row) {
      if (data.op !== "register") return json_({ cloud: 1, ok: false, err: "nouser" });
      row = last + 1;
      sh.getRange(row, 3).setNumberFormat("@");
      sh.getRange(row, 1, 1, 3).setValues([[text_(nick), text_(clean_(data.realName, 60)), code]]);
      sh.getRange(row, 11).setValue(new Date());
    } else {
      var saved = String(sh.getRange(row, 3).getDisplayValue()).trim();
      if (saved.length < 4) saved = ("0000" + saved).slice(-4);
      if (saved !== code) {
        cache.put(failKey, String(fails + 1), 900);
        return json_({ cloud: 1, ok: false, err: data.op === "register" ? "taken" : "code" });
      }
      // כינוי קיים עם הקוד הנכון: מחזירים את מה ששמור, והדף מאחד ושומר. לא דורסים
      if (data.op !== "save") {
        var raw = String(sh.getRange(row, 13).getValue() || "");
        var stored = null;
        try { stored = raw ? JSON.parse(raw) : null; } catch (err) {}
        return json_({ cloud: 1, ok: true, exists: 1, nick: sh.getRange(row, 1).getDisplayValue(), data: stored });
      }
    }

    var txt = JSON.stringify(data.data || {});
    if (txt.length > 45000) return json_({ cloud: 1, ok: false, err: "big" });
    var s = data.summary || {};
    if (clean_(data.realName, 60)) sh.getRange(row, 2).setValue(text_(clean_(data.realName, 60)));
    sh.getRange(row, 4, 1, 7).setValues([[text_(clean_(s.av, 8)), +s.score || 0, text_(clean_(s.level, 40)),
      +s.quests || 0, +s.powers || 0, +s.secrets || 0, +s.uploads || 0]]);
    sh.getRange(row, 12, 1, 2).setValues([[new Date(), txt]]);
    return json_({ cloud: 1, ok: true, nick: sh.getRange(row, 1).getDisplayValue() });
  } finally {
    lock.releaseLock();
  }
}

// ---------- קיר הרעיונות (ideas.html) ----------
var IDEA_HEAD = ["זמן", "שם (למורה)", "כיתה", "כינוי לתצוגה", "שם הרעיון", "מה הבעיה", "איך AI עוזר", "מסלול", "רוצה להשתתף", "מזהה", "סטטוס"];
var JOIN_HEAD = ["זמן", "שם", "כיתה", "מזהה רעיון", "שם הרעיון"];

function saveIdea_(ss, data) {
  var id = "i" + new Date().getTime().toString(36);
  var sh = sheetWithHeader_(ss, "רעיונות", IDEA_HEAD);
  sh.appendRow([
    new Date(), text_(clean_(data.realName, 60)), text_(clean_(data.cls, 20)), text_(clean_(data.nick, 30)),
    text_(clean_(data.title, 60)), text_(clean_(data.problem, 300)), text_(clean_(data.how, 300)),
    clean_(data.track, 40), data.wants ? "כן" : "", id, "חדש"
  ]);
  // מי שהציע ורוצה להשתתף נכנס גם לרשימת המשתתפים, על הרעיון שלו
  if (data.wants) saveJoin_(ss, { realName: data.realName, cls: data.cls, ideaId: id, ideaTitle: data.title });
  return ok_();
}

// "אני בפנים": מצטרפים לרעיון, או נרשמים להאקתון בלי רעיון (ideaId = "כללי")
function saveJoin_(ss, data) {
  var sh = sheetWithHeader_(ss, "הצטרפויות", JOIN_HEAD);
  sh.appendRow([new Date(), text_(clean_(data.realName, 60)), text_(clean_(data.cls, 20)),
    clean_(data.ideaId, 30) || "כללי", text_(clean_(data.ideaTitle, 60))]);
  return ok_();
}

// ---------- מה שמתפרסם באתר (רק מה שאושר) ----------
// ?op=public -> כוחות, משחקים ורעיונות מאושרים, בלי שמות אמיתיים
// ?op=game&id=N -> ה-HTML של משחק מאושר (N = מספר השורה בלשונית "ארקייד")
function doGet(e) {
  var p = (e && e.parameter) || {};
  if (p.op === "game") return json_(game_(p.id));
  var cache = CacheService.getScriptCache(), hit = cache.get("public");
  if (hit) return ContentService.createTextOutput(hit).setMimeType(ContentService.MimeType.JSON);
  var out = JSON.stringify(public_());
  try { cache.put("public", out, 60); } catch (err) {}
  return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON);
}

function approved_(v) {
  return /^\s*מאושר/.test(String(v || ""));
}

// כל שורות הלשונית כאובייקטים לפי שם העמודה, עם מספר השורה ב-_row
function table_(ss, name) {
  var sh = ss.getSheetByName(name);
  if (!sh || sh.getLastRow() < 2) return [];
  statusMenu_(sh);
  var v = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getValues(), head = v.shift();
  return v.map(function (r, i) {
    var o = { _row: i + 2 };
    head.forEach(function (h, j) { o[h] = r[j]; });
    return o;
  });
}

function public_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var powers = table_(ss, "כוחות").filter(function (r) { return approved_(r["סטטוס"]); }).map(function (r) {
    return { emoji: "⚡", name: String(r["שם הכוח"]), when: String(r["מתי משתמשים"]), spell: String(r["הלחש"]), by: String(r["כינוי"]) };
  });
  var games = table_(ss, "ארקייד").filter(function (r) { return approved_(r["סטטוס"]); }).map(function (r) {
    return { id: r._row, emoji: "🎮", title: String(r["שם המשחק"]), by: String(r["כינוי בארקייד"]), genre: String(r["ענף"]), proud: String(r["במה אני הכי גאה"]) };
  });
  var count = {}, joins = table_(ss, "הצטרפויות");
  joins.forEach(function (r) { var k = String(r["מזהה רעיון"]); count[k] = (count[k] || 0) + 1; });
  var ideas = table_(ss, "רעיונות").filter(function (r) { return approved_(r["סטטוס"]); }).map(function (r) {
    var id = String(r["מזהה"]);
    return { id: id, title: String(r["שם הרעיון"]), problem: String(r["מה הבעיה"]), how: String(r["איך AI עוזר"]),
      track: String(r["מסלול"]), by: String(r["כינוי לתצוגה"]), joins: count[id] || 0 };
  }).reverse();
  return { ok: true, powers: powers, games: games, ideas: ideas, joinsTotal: joins.length };
}

function game_(id) {
  var r = table_(SpreadsheetApp.getActiveSpreadsheet(), "ארקייד").filter(function (x) { return String(x._row) === String(id); })[0];
  if (!r || !approved_(r["סטטוס"])) return { ok: false };
  var m = String(r["קובץ"]).match(/[-\w]{25,}/);
  if (!m) return { ok: false };
  try { return { ok: true, html: DriveApp.getFileById(m[0]).getBlob().getDataAsString("UTF-8") }; }
  catch (err) { return { ok: false }; }
}
